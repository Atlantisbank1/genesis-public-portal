interface PrintableRecordResponse {
  status:
    'READY';

  record: {
    documentType:
      'STANDARD_TRUST_FORMATION_RECORD';

    documentVersion:
      '1.0';

    generatedAt:
      string;

    formation: {
      actionId:
        string;

      formationStatus:
        'COMPLETE';

      trustName:
        string | null;

      trustPurpose:
        string | null;

      settlorName:
        string | null;

      trusteeName:
        string | null;

      beneficiaryName:
        string | null;

      protectorName:
        string | null;

      initialPropertyDescription:
        string | null;

      formationCreatedAt:
        string;

      formationUpdatedAt:
        string;
    };

    membership: {
      memberId:
        string;

      status:
        string;

      subscriptionStatus:
        string;

      planCode:
        string;

      activatedAt:
        string | null;
    };

    serviceValidity: {
      status:
        'NOT_YET_ESTABLISHED';

      validFrom:
        null;

      validUntil:
        null;

      paymentStatus:
        null;

      paymentReference:
        null;

      paidAt:
        null;
    };
  };
}

function isPrintableRecordResponse(
  value:
    unknown,
): value is PrintableRecordResponse {
  if (
    typeof value !==
      'object' ||
    value ===
      null
  ) {
    return false;
  }

  if (
    !('status' in value) ||
    value.status !==
      'READY' ||
    !('record' in value)
  ) {
    return false;
  }

  const record =
    value.record;

  if (
    typeof record !==
      'object' ||
    record ===
      null
  ) {
    return false;
  }

  return (
    'documentType' in record &&
    record.documentType ===
      'STANDARD_TRUST_FORMATION_RECORD' &&
    'documentVersion' in record &&
    record.documentVersion ===
      '1.0' &&
    'formation' in record &&
    'membership' in record &&
    'serviceValidity' in record
  );
}

function displayValue(
  value:
    string | null,
): string {
  if (
    value ===
      null ||
    value.trim().length ===
      0
  ) {
    return 'Not recorded';
  }

  return value;
}

function safePdfFileName(
  value:
    string | null,
): string {
  const source =
    displayValue(value)
      .trim()
      .replace(
        /[^a-zA-Z0-9_-]+/g,
        '-',
      )
      .replace(
        /^[-_]+|[-_]+$/g,
        '',
      );

  return source.length > 0
    ? source
    : 'Standard-Trust';
}

          ).set(
            pdfBytes,
          );

          const blob =
            new Blob(
              [
                pdfArrayBuffer,
              ],
              {
                type:
                  'application/pdf',
              },
            );

          const objectUrl =
            URL.createObjectURL(
              blob,
            );

          const anchor =
            document.createElement(
              'a',
            );

          anchor.href =
            objectUrl;

          anchor.download =
            `${safePdfFileName(record.formation.trustName)}-Standard-Trust-Formation-Record.pdf`;

          document.body.appendChild(
            anchor,
          );

          anchor.click();

          anchor.remove();

          window.setTimeout(
            () => {
              URL.revokeObjectURL(
                objectUrl,
              );
            },
            1000,
          );
        }
        catch {
          setError(
            'The institutional PDF could not be generated.',
          );
        }
        finally {
          setGeneratingPdf(
            false,
          );
        }
      },
      [
        generatingPdf,
        record,
      ],
    );

  if (
    loading
  ) {
    return <PrintableRecordLoadingFallback />;
  }

    error !==
      null &&

      null
  ) {

            href="/trust-club/dashboard"

          >
            Return to Dashboard
          </Link>
        </section>
      </main>
    );
  }

  if (
    record ===
      null
  ) {
    return null;
  }

  return (

            </h1>

              Read-only record of the completed Standard Trust Formation.
            </p>
          </div>

          <div
            style={{

                'flex',

                'center',

                'wrap',
            }}
          >
            <button
              type="button"
              className="trustClubDashboardSignOut trustClubPrintablePrintButton"
              disabled={generatingPdf}
              onClick={() => {
                void generatePdf();
              }}
            >
              {generatingPdf
                ? 'Generating PDF...'
                : 'Generate PDF'}
            </button>

            <Link
              href="/trust-club/dashboard"
              className="trustClubDashboardSignOut trustClubPrintableDashboardLink"
            >
              Dashboard
            </Link>
          </div>
        </header>

        {error !== null && (
          <div className="trustClubError">
            {error}
          </div>
        )}

        <section className="trustClubAccessCard">
          <span className="trustClubAccessState trustClubAccessState-ACTIVE">
            COMPLETE
          </span>

          <h2>
            {displayValue(
              record.formation.trustName,
            )}
          </h2>

          <p>
            Document Type: {record.documentType}
          </p>

          <p>
            Document Version: {record.documentVersion}
          </p>

          <p>
            Action ID: {record.formation.actionId}
          </p>
        </section>

        <section className="trustClubDashboardGrid">
          <article className="trustClubDashboardCard">
            <span>

            </span>

            </small>

          <article className="trustClubDashboardCard">
